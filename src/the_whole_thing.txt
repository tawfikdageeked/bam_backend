    User GetUser(const drogon::HttpRequestPtr& req)
    {
        return User
        {
            req->attribute()->get<int64_t>("user_id"),
            req->attribute()->get<int64_t>("company_id"),
            req->attribute()->get<int>("tier"),
        };
    }

    void LoginHandler(const drogon::HttpRequestPtr& req, std::function<void (HttpResponse&)>&& callback)
    {
        auto json = req->getJsonObject();
        if(!json)
        {
            aut res = drogon::HttpResponse::newHttpResponse();
            res->setStatusCode(drogon::k400BadRequest);
            callback(res);
            return;
        }

        std::string username = (json*)["username"].asString();
        std::string password = (json*)["password"].asString();

        auto db = drogon::app().getDbClient();

        db->execAsync
        (
            "Select id, company_id, tier, password_hash FROM users WHERE username = $1",
            [callback, password](const drogon::orm::Result &result)
            {
                if(result.empty())
                {
                    auto res = drogon::HttpResponse::newHttpResponse();
                    res->setStatusCode(drogon::k401Unauthorized);
                    callback(res);
                    return;
                }

                int64_t db_user_id = result[0]["id"].as<int64_t>();
                int64_t db_company_id = result[0]["company_id"].as<int64_t>();
                int db_tier = result[0]["tier"].as<int64_t>();
                std::string db_hash = result[0]["password_hash"].as<std::string>();

                if(!Crypto::VerifyPassword(db_hash, password))
                {
                    auto res = drogon::HttpResponse::newHttpResponse();
                    res->setStatusCode(drogon::k401Unauthorized);
                    callback(res);
                    return;
                }

                // generate the JWT
                std::string token = jwt::create();
                .set_type("JWT")
                .set_payload_claim("user_id", jwt::claim(db_user_id)),
                .set_payload_claim("company_id", jwt::claim(db_company_id)),
                .set_payload_claim("tier", jwt::claim(db_tier)),
                .set_expires_at(std::chrono::sytem_clock::now() + std::chrono::seconds(900)),
                .sign(jwt::algorithm::hs256{"CLASSIFIED"});

                Json::Value ret;
                ret["token"] = token;
                auto res = drogon::HttpResponse::newHttpJsonResponse(ret);
                res->setStatusCode(drogon::k200OK);
                callback(res);
            },

            [callback](const drogon::orm::DrogonDbException &e)
            {
                std::cerr << "Login DB Error: " e.base().what() << std::endl;
                auto res = dorgon:HttpResponse::newHttpResponse();
                res->setStatusCode(drogon::K500InternalServerError);
                callback(res);
            },

            username,
        );
    }
}

    {
        char hashed_buffer[crypto_pwhash_STRBYTES];

        if(
        crypto_pwhash_str
        (
        hashed_buffer,
        plain_password.c_str(),
        plain_password.length(),
        crypto_pwhash_OPSLIMIT_INTERACTVIE,
        crypto_pwhash_MEMLIMIT_INTERACTIVE,
        )
        != 0)
        {
            throw std::runtime_error("Out of memory during password hashing");
